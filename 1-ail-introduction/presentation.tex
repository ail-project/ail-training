    \frametitle{Background}
    \begin{itemize}
        \item Over the past years, CIRCL has developed the AIL project\footnote{\url{https://www.ail-project.org/}} to fulfill our needs at CIRCL in intelligence gathering and analysis.

        \item The AIL Project is an \textbf{open-source} framework\footnote{\url{https://github.com/ail-project}} comprising various modules designed for the \textbf{collection, crawling, digging, and analysis of unstructured data}.

\section{Processing}


\begin{frame}{Processing Unsctrutured content}
	\begin{itemize}
		\item Collected data is often unstructured:
		\begin{itemize}
		    \item Free text, screenshots, documents, encoded data, \ldots
		\end{itemize}

		\item Relevant information is often embedded within the content:
		\begin{itemize}
		    \item Identifiers inside images or files
		    \item Obfuscated or partially formatted data
		\end{itemize}

		\item Direct search and correlation are unreliable without normalization

	\end{itemize}
\end{frame}

\begin{frame}{Creating a YARA Rule}
	\textbf{What is YARA?}
	\begin{itemize}
		\item An improved \texttt{grep} for pattern matching
		\item Searches for strings, binary patterns, and regular expressions
		\item Rules can include metadata and tags for context
		\item Easier and more structured conditions than complex regex
	\end{itemize}

	\vspace{0.3cm}
	\textbf{Basic Structure of a YARA Rule}

	\begin{itemize}
		\item \textbf{rule} my\_new\_rule\_name
		\item \textbf{meta:} descriptive information
		\item \textbf{strings:} patterns to search for
		\item \textbf{condition:} logical expression defining when it matches
	\end{itemize}
\end{frame}

\begin{frame}[fragile]{YARA Rule Examples - Tracking a specific mail domain}
\begin{verbatim}
rule mail_proton
{
    meta:
        description = "Search for a specific mail domain name"

    strings:
        $a = "@protonmail.com" nocase

    condition:
        $a
}
\end{verbatim}
\end{frame}

\begin{frame}[fragile]{YARA Rule Examples - Detecting Disney+ credential leaks}
\begin{verbatim}
rule disney_plus
{
    meta:
        description = "Finding list of credentials for Disney Plus"

    strings:
        $a = "gmail.com:" nocase
        $b = "DISNEY_PLUS"
        $c = "Disney Plus"

    condition:
        $a and ($b or $c)
}
\end{verbatim}
\end{frame}

\section{Search}

\begin{frame}
    \frametitle{Query Operators}
    \begin{itemize}
        \item Use double quotes \texttt{"word"} for \textbf{exact matches} (disables fuzzy/typo matching)
  		\item Use \texttt{-word} to \textbf{exclude terms} from results
  		\item Use dot \texttt{.} as a \textbf{single-character wildcard}
    \end{itemize}
    \begin{figure}
        \includegraphics[scale=0.22]{screenshot/search.png}
    \end{figure}
\end{frame}

\begin{frame}[t]
	\frametitle{Search Scope}
	\begin{columns}[T,onlytextwidth]
        \column{0.45\textwidth}
            \centering
            % Enlarge sidebar labels for projection; allow overflow below the slide.
            \includegraphics[height=1.20\textheight, keepaspectratio]{screenshot/search_scopes.png}
        \column{0.55\textwidth}
            \begin{itemize}
		\item Use the \textbf{left sidebar} to select what the query should search.
	  	\item Default: \textbf{All} sources (best for general discovery).
	  	\item Narrow scope to reduce noise and improve relevance:
		\begin{itemize}
		  \item \textbf{Chats}: Discord, Matrix, Telegram
		  \item \textbf{Crawled domains}: Web, Tor (.onion), I2P
		  \item \textbf{HTML titles}
		  \item \textbf{Filenames}
		  \item \textbf{AI descriptions}: Images, Screenshots, Domains
		\end{itemize}

    \end{columns}
\end{frame}

\begin{frame}
    \frametitle{Result Ordering and Time Filters}
    \begin{itemize}
    	\item \textbf{Most Recent (default)}: Orders results by most recently seen objects first.
		\item \textbf{Recent}: Prioritizes results with stronger matches and higher occurrence counts.
		\item \textbf{Last Seen} Limits results to objects \textbf{last observed} within a specified date range.
	\end{itemize}
